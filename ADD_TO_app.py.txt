@app.route('/api/market-news')
def get_market_news():
    """Fetch live market news (JSON, used by market_news.html via JS)"""
    try:
        force_refresh = request.args.get('refresh', '0') == '1'
        limit = request.args.get('limit', 30, type=int)

        news_items = news_fetcher.fetch_all_news(limit=limit, force_refresh=force_refresh)

        # Make datetimes JSON-safe
        serializable_news = []
        for item in news_items:
            item_copy = dict(item)
            item_copy.pop('published_parsed', None)
            serializable_news.append(item_copy)

        return jsonify({
            'status': 'success',
            'count': len(serializable_news),
            'news': serializable_news,
            'timestamp': datetime.now().isoformat()
        })

    except Exception as e:
        logger.error(f"Error fetching market news: {str(e)}", exc_info=True)
        return jsonify({'error': f'Error fetching market news: {str(e)}'}), 500
